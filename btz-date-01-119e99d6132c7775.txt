    cache.set(key, result); return result;
  }
  function matches(sku, value, filters = {}) {
    const mode = normalizeValue(value);
    if (!mode) return true;
    const state = flags(sku, filters);
    return mode === 'down' ? state.hasDown : mode === 'up' ? state.hasUp : state.hasDown && state.hasUp;
  }
