/* 在原销量模型首次初始化前恢复本次真取数的新快照；旧来源资产留作审计。 */
(function (root) {
  'use strict';
  const element = document.getElementById('monthTodaySalesDelta');
  if (!element) throw Error('缺少本次销量增量');
  const delta = JSON.parse(element.textContent);
  if (delta.schema !== 'muku.sales-sequence-delta.v1') throw Error('销量增量格式不支持');
  root.monthTodaySalesPacketProvider = function (text) {
    const original = JSON.parse(text), output = { ...delta.values }, copied = {}, inserted = {};
    for (const [key, operations] of Object.entries(delta.arrays)) {
      const previous = original[key];
      if (!Array.isArray(previous) || previous.length !== delta.baseCounts[key]) throw Error('旧来源数量不符：' + key);
      const rows = []; copied[key] = 0; inserted[key] = 0;
      for (const operation of operations) {
        if (operation.copy) {
          const [start, length] = operation.copy;
          if (!Number.isInteger(start) || !Number.isInteger(length) || start < 0 || length < 1 || start + length > previous.length) throw Error('销量复制范围不符：' + key);
          rows.push(...previous.slice(start, start + length)); copied[key] += length;
        } else if (Array.isArray(operation.items)) {
          rows.push(...operation.items); inserted[key] += operation.items.length;
        } else throw Error('销量增量操作不支持：' + key);
      }
      if (rows.length !== delta.targetCounts[key]) throw Error('新来源数量不符：' + key);
      output[key] = rows;
    }
    const expected = [...delta.targetKeys].sort().join('|');
    if (Object.keys(output).sort().join('|') !== expected) throw Error('新来源字段不符');
    root.monthTodaySourceModel = { ready: true, baseSha256: delta.baseSha256, targetSha256: delta.targetSha256, copied, inserted, meta: output.meta };
    return output;
  };
})(window);

<script>
/* 日期快捷入口按美西业务今天取值，来源覆盖仍由销量模型判断。 */
(function (root) {
  'use strict';
  if (root.monthTodayUiModel?.ready) return;
  if (!root.salesRefreshModel?.runtimeReady) throw Error('月初至今天快捷入口需要销量模型');
  const timeZone = 'America/Los_Angeles';
  const formatter = new Intl.DateTimeFormat('en-CA', { timeZone, year: 'numeric', month: '2-digit', day: '2-digit' });
  let selected = '';
  function todayRange(instant = new Date()) {
    const fields = Object.fromEntries(formatter.formatToParts(new Date(instant)).map(part => [part.type, part.value]));
    const end = fields.year + '-' + fields.month + '-' + fields.day;
    return { start: end.slice(0, 7) + '-01', end };
  }
  function presetRange(key, dataset) {
    if (key === 'monthToday') return todayRange();
    const coverage = root.salesRefreshModel.getCoverage(key === 'year' ? 'annual' : dataset);
    return key === 'year' ? { start: coverage.start.slice(0, 4) + '-01-01', end: coverage.start.slice(0, 4) + '-12-31', dataset: 'annual' }
      : { start: sxAdd(coverage.end, 1 - Number(key)), end: coverage.end };
  }
  function read() {
    return { start: cxEl('sxDateStart')?.value, end: cxEl('sxDateEnd')?.value, dataset: cxEl('sxDateDataset')?.value };
  }
  function matches(key, range) {
    const expected = presetRange(key, range.dataset);
    return (!expected.dataset || range.dataset === expected.dataset) && range.start === expected.start && range.end === expected.end;
  }
  function sync() {
    const dialog = cxEl('sxDateDialog'), buttons = [...dialog?.querySelectorAll('[data-mtt-preset]') || []];
    if (!buttons.length) return;
    const range = read(), keys = buttons.map(button => button.dataset.mttPreset);
    if (!keys.includes(selected) || !matches(selected, range)) selected = keys.find(key => matches(key, range)) || '';
    for (const button of buttons) {
      const pressed = String(button.dataset.mttPreset === selected);
      if (button.getAttribute('aria-pressed') !== pressed) button.setAttribute('aria-pressed', pressed);
      button.classList.add('fa-state-control', 'mtt-state-control');
      if (button.dataset.mttPreset === 'monthToday') {
        const today = todayRange();
        button.title = '美西业务日期（America/Los_Angeles）：' + today.start + '—' + today.end + '；未覆盖日期保留待核';
      }
    }
  }
  function install() {
    const dialog = cxEl('sxDateDialog'), host = dialog?.querySelector('.sx-date-presets');
    if (!host) return;
    // 旧快捷监听会停止事件传播；只转换弹窗按钮属性，统一管理其互斥选中态。
    for (const button of host.querySelectorAll('[data-srf-preset],[data-sx-preset]')) {
      button.dataset.mttPreset = button.dataset.srfPreset || button.dataset.sxPreset;
      button.removeAttribute('data-srf-preset'); button.removeAttribute('data-sx-preset');
    }
    if (!host.querySelector('[data-mtt-preset="monthToday"]')) {
      const button = document.createElement('button'); button.type = 'button'; button.dataset.mttPreset = 'monthToday';
      button.textContent = '月初至今天'; host.prepend(button);
    }
    sync();
  }
  const priorDialog = sxDateDialog;
  sxDateDialog = function () { const result = priorDialog(); install(); return result; };
  root.addEventListener('click', event => {
    const button = event.target.closest('#sxDateDialog [data-mtt-preset]'); if (!button) return;
    event.preventDefault(); event.stopImmediatePropagation();
    selected = button.dataset.mttPreset;
    const dataset = cxEl('sxDateDataset'), range = presetRange(selected, dataset.value);
    if (range.dataset) dataset.value = range.dataset;
    cxEl('sxDateStart').value = range.start; cxEl('sxDateEnd').value = range.end;
    cxEl('sxMonth').value = range.end.slice(0, 7);
    cxEl('sxDateError').textContent = '';
    sxDateHint(); sync();
  }, true);
  function manual(event) {
    if (!['sxDateStart', 'sxDateEnd', 'sxDateDataset', 'sxMonth'].includes(event.target.id)) return;
    selected = '';
    Promise.resolve().then(sync);
  }
  document.addEventListener('input', manual, true); document.addEventListener('change', manual, true);
  root.monthTodayUiModel = { ready: true, timeZone, todayRange, presetRange, matches, install, sync };
})(window);

</script>
