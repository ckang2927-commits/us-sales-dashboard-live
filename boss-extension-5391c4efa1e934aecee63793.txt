        '与SKU数据查询共用同SKU最近最多20笔有效成本订单，先按平台店铺筛选，再按收入加权。通知价预估与原成交价模式分别标记；日期和费用模型在详情单列，未含累计仓储。此图不是所选期间实际总利润。');
    return { ...result, summary: result.summary.replaceAll('现价', '通知价'), html: html
      + '<p class="cx-note opb-chart-note">所有利润分层和关注清单与查询主列共用同一利润模型；本轮新增补证标为自身成本预估，来源与费用假设可在SKU“利润与调价”中展开。未知成本不作0。</p>' };
  };
  const priorCell = sqUiCell;
  sqUiCell = function (column, row) {
    const html = priorCell(column, row), profit = row._profit20;
    if (!profit?.ownBackfill) return html;
    if (column === 'operatingMargin' || column === 'realMargin') return html.replace(/最近\d+单综合/g, '同SKU自身成本预估');
    if (column === 'evidence') return html.replace('pp-status ', 'pp-status estimate ')
      + '<small class="pp-sub opb-missing">' + esc(profit.selected ? '原采购与费用来源已匹配；估算项逐单披露'
        : (profit.ownBackfillEvidence.missingFields || []).join('；')) + '</small>';
    if (column === 'scope') return '<small class="pp-sub">报价原引用</small>'
      + html.replace('利润参考仓：', '报价原参考仓：')
      + '<small class="pp-sub">利润样本来源：' + esc(profit.selected?.stores?.join('；') || '当前平台／店铺无可用自身样本') + '</small>';
    return html;
  };
  function evidenceHtml(sku, filters) {
    const result = model.forSku(sku, filters), e = result.ownBackfillEvidence;
    if (!e) return '';
    const samples = result.samples || [];
    const notes = [...new Set([...(e.notes || []), ...samples.flatMap(row => row.notes || [])])];
    return '<section class="pp-section opb-evidence"><h3>同SKU历史订单与采购补证</h3>'
      + '<p class="pp-note">' + esc(result.selected ? '本次补齐采用本SKU自己的历史销售、采购与费用依据；费用模型和当前仓储模拟均单独标注。此处为成本预估，不是财务实结利润。'
        : '已检索本SKU自己的历史订单、采购登记及物流账单；现有证据尚不能闭合全部必要成本，不使用其他SKU补值。') + '</p>'
      + '<details><summary>查看本次核对范围、估算口径与剩余缺项</summary>'
      + '<p>' + esc('自身历史候选 ' + (e.candidateCount || 0) + '单；有可用成本依据 '
        + result.validSampleCount + '单。按所选平台／店铺先筛后取最近最多20单，销量日期不截断成本历史。') + '</p>'
      + (!result.selected ? '<p><strong>仍缺：</strong>' + esc((e.missingFields || []).join('；')) + '</p>' : '')
      + notes.map(note => '<p>' + esc(note) + '</p>').join('')
      + '<p>运营采购＝原真实采购×1.5；两种利润采用同一收入、其他费用及售后储备分别重算。缺失费用不按0，重复账单和未明确分摊的混合订单不强行计入。</p>'
      + '</details></section>';
  }
  function currentFilters() {
    return cxState.view === 'query' ? sqUiState.filters
      : { ...cfReferenceScope(), evidenceMode: cfState.profitMode === 'current' ? 'current' : 'withHistory' };
  }
  function sampleEvidenceHtml(sample) {
    const purchase = sample.purchaseEvidence || {}, tail = sample.tailEvidence || {}, volume = sample.volumeEvidence || {};
    const money = value => finite(value) ? cxMoney(value) : '未取得';
    const reference = [tail.sourceOrderId || sample.orderId, tail.referenceDate || '',
      tail.platform ? (tail.platform === 'Shopify' ? '独立站' : tail.platform) : '', tail.store || ''].filter(Boolean).join(' · ');
    const lines = [
      '采购原价：' + (purchase.trueCnyUnit ?? '未取得') + ' ' + (purchase.purchaseCurrency || sample.sourcePurchaseCurrency || '')
        + '/件；采购参考日期：' + (purchase.referenceDates?.join('、') || '原记录日期未核')
        + '；汇率参数：1 USD＝' + (purchase.modelFx ?? sample.sourcePurchaseFxCnyPerUsd ?? '未核') + ' CNY（非实付汇率）。',
      '尾程' + (tail.actual ? '本单费用依据：' : '估算参考：') + reference
        + (finite(tail.tailPerUnit) ? '；参考每件 ' + money(tail.tailPerUnit) : '')
        + '；本次计入 ' + money(sample.componentTotals.tail) + (tail.actual ? '。' : '，不称本单实际尾程。'),
      '自身总体积：' + (finite(volume.totalCbm) ? Number(volume.totalCbm.toFixed(6)) + ' m³' : '未取得')
        + (volume.basis === 'same_order_complete_packages' ? '；同订单完整包件。' : '；同SKU自身尺寸参考，非本单实际测量。'),
      '作业费模型 ' + money(sample.handlingModelTotal) + '；已证出库作业 ' + money(sample.actualOutboundHandlingControlTotal)
        + '；本次计入 ' + money(sample.componentTotals.handling) + '，采用两者较高值，未重复相加。'
    ];
    return '<div class="opb-sample-notes">' + [...lines, ...(sample.notes || [])].map(note => '<p>' + esc(note) + '</p>').join('') + '</div>';
  }
  const priorDetail = cxRenderSku;
  cxRenderSku = function () {
    // 旧详情渲染器仅认顶层平台/店铺，且写死current；同步本次视图上下文，完成后原样还原。
    const filters = currentFilters(), isSalesDetail = cxState.view !== 'query';
    const platform = cxState.sales.platform, store = sxState.store, forSku = model.forSku;
    let result;
    try {
      if (isSalesDetail) {
        cxState.sales.platform = filters.platform || ''; sxState.store = filters.store || '';
        model.forSku = (id, scope) => forSku(id, { ...scope, ...filters });
      }
      result = priorDetail.apply(this, arguments);
    } finally {
      if (isSalesDetail) { cxState.sales.platform = platform; sxState.store = store; model.forSku = forSku; }
    }
    const sku = cxState.detail?.sku;
    const body = cxEl('cxDetailBody');
    if (!sku || !records.has(sku) || !body?.querySelector('.pp-section')) return result;
    if (!body.querySelector('.opb-evidence')) body.insertAdjacentHTML('afterbegin', evidenceHtml(sku, currentFilters()));
    // 将该订单的估算假设紧跟已有费用及原始来源，避免在主表堆长说明。
    const profit = model.forSku(sku, currentFilters());
    if (profit.ownBackfill) for (const row of body.querySelectorAll('.pp-samples-wrap tbody tr')) {
      const orderId = row.cells?.[1]?.childNodes?.[0]?.textContent?.trim();
      const sample = profit.samples.find(sample => orderId === sample.orderId);
      const detail = row.querySelector('details');
      if (sample && detail && !detail.querySelector('.opb-sample-notes')) detail.insertAdjacentHTML('beforeend',
        sampleEvidenceHtml(sample));
    }
    return result;
  };
  const currentCsvName = name => name.replace(/_\d{8}(?=\.csv$|$)/, '_' + packet.asOf.replaceAll('-', ''));
  // 原库存补丁会在CSV链深处把文件名改回11/16；在实际保存入口统一本次日期。
  const priorManualDownload = downloadManual;
  downloadManual = function (value, name, type) {
    return priorManualDownload(value, String(name).startsWith('SKU数据查询_') ? currentCsvName(name) : name, type);
  };
  const priorExport = sqUiExport;
  sqUiExport = function () {
    const rows = sqExportRows({ ...sqUiState.filters }, { ...sqUiState.sort }), bySku = new Map(rows.map(row => [row.sku, row]));
    const priorDownload = cxDownloadCsv;
    cxDownloadCsv = function (name, headers, values) {
      if (!String(name).startsWith('SKU数据查询_')) return priorDownload(name, headers, values);
      const skuColumn = headers.indexOf('商品SKU');
      if (skuColumn < 0) throw Error('自身利润补证导出缺少SKU身份列');
      return priorDownload(currentCsvName(name),
        [...headers, '本轮自身利润补证', '自身补证缺项', '自身费用估算口径'], values.map(value => {
        const row = bySku.get(value[skuColumn]);
        if (!row) throw Error('自身利润导出与查询人口不一致');
        return [...value, row.profitBackfillStatus || '', row.profitMissingFields || '', row.profitEstimationNotes || ''];
      }));
    };
    try { return priorExport(); } finally { cxDownloadCsv = priorDownload; }
  };
  model.clearCaches(); root.spQueryPeriodModel?.clearCaches(); ldReferenceCache.clear();
  root.allSkuOwnProfitModel = { ready: true, asOf: packet.asOf, records, samplesBySku,
    summary: packet.summary, evidenceHtml, forSku: (sku, filters) => model.forSku(sku, filters) };
  if (cxState.view === 'query') sqRenderQuery();
  else if (cxState.view === 'sales') cxRenderSales();
})(window);
