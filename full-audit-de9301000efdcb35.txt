      const key = attr(attributes, action), series = attr(attributes, 'data-sx-series');
      if (model.hidden.has(key + '|' + series) && ['sx-trend-point', 'sx-trend-point-label', 'sx-trend-point-guide', 'sx-trend-bar', 'sx-trend-segment-label'].some(value => hasClass(attributes, value))) return '';
      if (model.hiddenTotals.has(key) && hasClass(attributes, 'sx-trend-total-label')) return '';
      if (hasClass(attributes, 'sx-trend-line-hit') && (model.hidden.has(key + '|' + series) || model.hidden.has(keyForX(attr(attributes, 'x1')) + '|' + series))) return '';
      return tag;
    });
    html = html.replace(/<polyline\b([^>]*)>[\s\S]*?<\/polyline>/g, (tag, attributes) => {
      if (!hasClass(attributes, 'sx-trend-line')) return tag;

        .map(points => tag.replace(/points="[^"]*"/, 'points="' + points.join(' ') + '"')).join('');
    });
    html = html.replace(/<g\b([^>]*)>([\s\S]*?)<\/g>/g, (tag, attributes, content) => {
      if (!hasClass(attributes, 'sx-trend-tracer')) return tag;

      if (!motion) return tag;
      return splitPoints(motion[1].replace(/^M/, '').split(/\s+L/), attr(attributes, 'data-sx-series'))
        .map(points => tag.replace(/path="[^"]*"/, 'path="M' + points.join(' L') + '"')).join('');
    });
    html = html.replace(/(<g class="sx-trend-grid[^"]*"[^>]*>[\s\S]*?<text\b[^>]*>)([^<]*)(<\/text>)/g,
      (tag, start, value, end) => start + esc(formatNumber(value, measure, true)) + end);
    html = html.replace(/<text\b([^>]*)>([\s\S]*?)<\/text>/g, (tag, attributes, content) => {

      const formatted = content.replace(/(-?[\d,]+(?:\.\d+)?)$/, value => esc(formatNumber(value, measure, measure === 'amount')));
      return '<text' + attributes + '>' + (measure === 'share' && hasClass(attributes, 'sx-trend-total-label')
        ? '<title>平台商品销售额占比合计；单位百分比</title>' : '') + formatted + '</text>';
    });
    const metricText = content => {
      for (const group of options.groups || sxPlatforms) {
        const label = String(group.label || group.key).replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        content = content.replace(new RegExp('(' + label + ') (-?[\\d,]+(?:\\.\\d+)?)(?:套|件)?(?![\\d.%])', 'g'),
          (match, name, value) => name + ' ' + formatNumber(value, measure));
      }
      content = content.replace(/合计 (-?[\d,]+(?:\.\d+)?)(?:套|件)?(?![\d.%])/g, (match, value) =>
        (measure === 'share' ? '平台占比合计 ' : '商品额合计 ') + formatNumber(value, measure));
      content = content.replace(/(?:已核净数量（下界）|历史订单参考数量|销量（原订单数量）)/g, measures[measure]);
      return content + ' · ' + measures[measure];

    html = html.replace(/<title>([\s\S]*?)<\/title>/g, (tag, content) => '<title>' + metricText(content) + '</title>');
    html = html.replace(/aria-label="([^"]*)"/g, (tag, content) => 'aria-label="' + metricText(content) + '"');
    html = html.replace(/<g\b([^>]*class="sx-trend-bucket[^>]*>)([\s\S]*?)<\/g>/g, (tag, attributes, content) => {
      const bucket = model.map.get(attr(attributes, action));
      if (!bucket || !model.hiddenTotals.has(bucket.key)) return tag;
      const note = bucket.label + '；' + (bucket.missing ? '金额待补' : bucket.negative ? '含负金额，不计算普通占比' : '无正销售额，不计算占比');
      return '<g' + attributes.replace(/aria-label="[^"]*"/, 'aria-label="' + esc(note) + '"') + content.replace(/<title>[\s\S]*?<\/title>/, '<title>' + esc(note) + '</title>')
        .replace(/<line class="sx-trend-empty-bar"[^>]*>[\s\S]*?<\/line>/g, '') + '</g>';
