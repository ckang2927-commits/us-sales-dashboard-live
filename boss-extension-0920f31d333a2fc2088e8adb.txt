  function operationsText() {
    const visible = filtered();
    const queue = queueDefinitions.find(item => item.id === state.queue);
    const lines = ['@所有人 各位运营同事，以下为' + date(asOf) + '降价复核结论（' + queue.label + '，' + visible.length + '款）：'];
    if (queue.description) lines.push(queue.description);
    const classification = row => {
      const tags = ruleLabels(row, verifiedReductions(row));
      return tags.length ? ' [' + tags.join(' / ') + ']' : '';
    };
    const actionable = visible.filter(row => ['candidate','history'].includes(actionStatus(row).code));
    const withheld = visible.filter(row => !actionable.includes(row));
    if (!actionable.length) lines.push('当前筛选没有可列入处理建议的SKU。');
    const actionOrder = ['second_round', 'wayfair', 'quote_review', 'scope_review', 'sales_review', 'observe_period', 'executed_observe', 'gray_monitor', 'arrival_review', 'normal_review'];
    const actions = [...new Set([...actionOrder, ...actionable.map(row => row.action)])];
    for (const action of actions) {
      const group = actionable.filter(row => row.action === action);
      if (!group.length) continue;
      lines.push('', (actionLabels[action] || group[0].actionLabel || '复核') + '（' + group.length + '款）');
      for (const row of group) lines.push(row.sku + classification(row) + '：' + label(row) + '。' + (row.reason || '') + (row.nextReviewDate ? ' ' + (row.conditionalDates ? '条件复核日' : '复核日') + date(row.nextReviewDate) + '。' : ''));
    }
    if (withheld.length) {
      lines.push('', '以下仅供重核或暂停记录，暂不发起改价或转仓。');
      for (const row of withheld) { const item=actionStatus(row);lines.push(row.sku+classification(row)+' · '+item.label+'。'+item.reason); }
    }
    lines.push('', '红色已于11月17日降价，绿色为已降，均避免重复同一轮折扣。清仓策略：先20%，观察15天，再减原价10%至累计30%，后续仍卖不出再转Wayfair。', '销量证据截至' + date(meta.salesThrough) + '；未覆盖的调后日期待补，不按0销量判断。');
    lines.push('以上为看板复核信息。是否改价或转仓，以逐店核准后实际生效为准。');
    return lines.join('\n');
  }

      setMessage('已复制当前筛选' + filtered().length + '款的运营文案。');
      return true;
    } catch (_) {
      const area = el('mdrCopyText'); area.value = content; el('mdrCopyFallback').hidden = false; area.focus(); area.select();
      try { if (document.execCommand('copy')) { setMessage('已复制当前筛选运营文案。'); return true; } } catch (_) {}
      setMessage('文案已展开，可选中后复制。'); return false;
    }
  }
