window.createRefundAudit = function (refundRows, platformRows) {
 'use strict';
 const cents = value => typeof value === 'number' && Number.isFinite(value) ? Math.round(value*100) : null;
 const keyFor = row => JSON.stringify([row.platform,row.store,row.orderId,row.refundCompletedDate,cents(row.actualRefundAmount),row.typeLabel,row.description]);
 const active = row => (!row.disposition || row.disposition === 'kept') && row.includeInMetrics !== false;
 const sourceKeys = row => {
  const source = row.source || {};
  return [...new Set(source.rows?.length ? source.rows : [source.row])]
   .filter(value => value != null)
   .map(value => JSON.stringify([source.file || source.path || '',source.sheet || '',value]));
 };
 const itemKeys = row => new Set([row.inventorySku,row.canonicalSku,row.sourceSku,...(row.skus||[])].filter(Boolean));
 const eventsById = new Map(platformRows.map(row => [row.id,row]));
 const groupsByKey = new Map();
 for (const row of refundRows) {
  if (!active(row) || !row.orderKey || !row.orderId || !row.refundCompletedDate || !(cents(row.actualRefundAmount)>0)) continue;
  const key = keyFor(row);
  if (!groupsByKey.has(key)) groupsByKey.set(key,[]);
  groupsByKey.get(key).push(row);
 }
 const groupsById = new Map();
 for (const group of groupsByKey.values()) {
  if (group.length < 2) continue;
  const matched = group.map(row => {
   const keys = itemKeys(row);
   const events = [...new Set(row.platformEvidenceRefs || [])].map(id => eventsById.get(id)).filter(event =>
    event && active(event) && event.orderKey === row.orderKey && event.kind === 'refund_billing'
    && cents(event.bookRefund) === cents(row.actualRefundAmount)
    && [...itemKeys(event)].some(sku => keys.has(sku)) && sourceKeys(event).length);
   return events.length === 1 ? events[0] : null;
  });
  const eventIds = new Set(matched.map(row => row?.id));
  const allSourceKeys = matched.flatMap(row => row ? sourceKeys(row) : []);
  const independent = matched.every(Boolean) && eventIds.size === group.length
   && allSourceKeys.length === new Set(allSourceKeys).size;
  group.forEach((row,index) => groupsById.set(row.id,{independent,event:independent ? matched[index] : null}));
 }
 function flags(row) {
  if (!row.id || !row.id.startsWith('RR-')) return [];
  const list = [];
  const over = cents(row.amount) != null && cents(row.orderAmount) != null && cents(row.amount) > cents(row.orderAmount)+2;
  if (over) list.push('退款登记高于订单基数');
  const ids = new Set((row.skuMatches || []).map(item => item.sku).filter(Boolean));
  if (!ids.size) (row.skus || []).forEach(sku => ids.add(sku));
  if (ids.size > 1) list.push('多SKU整单金额未分摊');
  if (!row.refundCompletedDate) list.push('退款完成日待核');
  const item = groupsById.get(row.id);
  if (item) list.push(item.independent ? '同额分录独立' : '同额记录待排重');
  return list;
 }
 return {flags, groupInfo: id => groupsById.get(id)};
};
